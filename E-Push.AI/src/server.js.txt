require('dotenv').config();
const express = require('express');
const helmet = require('helmet');
const rateLimit = require('express-rate-limit');
const cors = require('cors');
const NodeCache = require('node-cache');

const botDetector = require('./middleware/botDetector');
const UniversalParser = require('./services/parser');
const Normalizer = require('./services/normalizer');
const LLMsGenerator = require('./services/llmsGenerator');

const app = express();
const cache = new NodeCache({ stdTTL: process.env.CACHE_TTL_SECONDS || 3600 });

app.use(helmet());
app.use(cors({ origin: process.env.ALLOWED_ORIGIN || '*' }));
app.use(express.json());
app.use(botDetector);

const limiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 300,
  message: { error: 'Too many requests, try again later.' }
});
app.use(limiter);

const partnerConfig = {
  name: process.env.PARTNER_NAME,
  city: process.env.PARTNER_CITY,
  phone: process.env.PARTNER_PHONE,
  whatsapp: process.env.PARTNER_WHATSAPP,
  instagram: process.env.PARTNER_INSTAGRAM,
  tapaz: process.env.PARTNER_TAPAZ,
  address: process.env.PARTNER_ADDRESS,
  geoLat: process.env.PARTNER_GEO_LAT,
  geoLon: process.env.PARTNER_GEO_LON,
  birkartMaxMonths: parseInt(process.env.PARTNER_BIRKART_MAX_MONTHS) || 12
};

const mockRawProducts = [
  { id: "1", title: "Apple iPhone 16 Pro 128GB Black Titanium", price: 2150, oldPrice: 2290, inStock: true },
  { id: "2", title: "Sony PlayStation 5 Slim 1TB (Disc Edition)", price: 1090, oldPrice: 1200, inStock: true },
  { id: "3", title: "Xiaomi Robot Vacuum S10+", price: 650, oldPrice: null, inStock: true }
];

app.get('/llms.txt', async (req, res) => {
  try {
    const cachedLLM = cache.get('llms_content');
    if (cachedLLM) {
      res.setHeader('Content-Type', 'text/plain; charset=utf-8');
      return res.send(cachedLLM);
    }

    const parsedData = await UniversalParser.parse(mockRawProducts, 'json');
    const processedProducts = parsedData.map(item => Normalizer.processProduct(item, partnerConfig));
    const llmsContent = LLMsGenerator.generate(processedProducts, partnerConfig);

    cache.set('llms_content', llmsContent);

    res.setHeader('Content-Type', 'text/plain; charset=utf-8');
    res.send(llmsContent);
  } catch (error) {
    console.error('[E-PUSH.AI ERROR]', error);
    res.status(500).send('# Error generating llms.txt');
  }
});

app.post('/api/v1/update-catalog', async (req, res) => {
  const { rawData, format } = req.body;
  if (!rawData) {
    return res.status(400).json({ error: 'Data is required' });
  }

  try {
    const parsedData = await UniversalParser.parse(rawData, format || 'json');
    const processedProducts = parsedData.map(item => Normalizer.processProduct(item, partnerConfig));
    const newLLMsContent = LLMsGenerator.generate(processedProducts, partnerConfig);

    cache.set('llms_content', newLLMsContent);

    res.json({
      status: 'success',
      count: processedProducts.length,
      message: 'Catalog updated and llms.txt regenerated'
    });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

app.get('/health', (req, res) => {
  res.json({ status: 'ok', service: 'E-Push.AI', timestamp: new Date().toISOString() });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`================================================`);
  console.log(`[E-Push.AI Engine Online]`);
  console.log(`Server listening on port: ${PORT}`);
  console.log(`Endpoint: http://localhost:${PORT}/llms.txt`);
  console.log(`================================================`);
});
